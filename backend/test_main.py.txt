from fastapi.testclient import TestClient
from main import app

# Create a test client using your FastAPI app
client = TestClient(app)

def test_health_check_returns_200():
    # Simulate a GET request to your health endpoint
    response = client.get("/api/health")
    
    # Verify the status code and the exact JSON response
    assert response.status_code == 200
    assert response.json() == {"status": "healthy", "message": "API is running securely"}

def test_invalid_endpoint_returns_404():
    # Simulate a request to an endpoint that doesn't exist
    response = client.get("/api/this-does-not-exist")
    
    # Verify the backend correctly identifies it as Not Found
    assert response.status_code == 404
